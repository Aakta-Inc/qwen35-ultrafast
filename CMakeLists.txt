cmake_minimum_required(VERSION 3.24)
project(qwen35_ultrafast LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_library(qwen35_kernels STATIC cuda/gated_rmsnorm.cu)
target_include_directories(qwen35_kernels PUBLIC cuda)
add_executable(qwen35-kernel-bench cuda/kernel_bench.cu)
target_link_libraries(qwen35-kernel-bench PRIVATE qwen35_kernels)
enable_testing()
add_test(NAME gated_rmsnorm_correctness_and_benchmark COMMAND qwen35-kernel-bench)
